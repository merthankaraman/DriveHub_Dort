    private void initMaintenanceInfo() {

        LogUtil.logD(TAG, "initMaintenanceInfo: " + maintenanceInfo);
        if (maintenanceInfo != null && !maintenanceInfo.isEmpty()) {
            int resetMileage = Integer.parseInt((String) Objects.requireNonNull(maintenanceInfo.get(KEY_MAP_RESET_MILEAGE)));
            long resetDate = Long.parseLong((String) Objects.requireNonNull(maintenanceInfo.get(KEY_MAP_RESET_DATE)));
            this.mResetDate = resetDate;
            this.mResetMileage = resetMileage;
            this.mStdMtcMileage = Integer.parseInt((String) Objects.requireNonNull(maintenanceInfo.get(KEY_MAP_STD_MILEAGE)));
            this.mStdMtcTime = Integer.parseInt((String) Objects.requireNonNull(maintenanceInfo.get(KEY_MAP_STD_TIME)));
            this.mStdMtcAdvMileage = Integer.parseInt((String) Objects.requireNonNull(maintenanceInfo.get(KEY_MAP_STD_ADV_MILEAGE)));
            this.mStdMtcAdvTime = Integer.parseInt((String) Objects.requireNonNull(maintenanceInfo.get(KEY_MAP_STD_ADV_TIME)));
            if (this.mResetDate == 0) {
                TspManager.getInstance().getMaintenanceInfo();
            } else {
                TspManager.getInstance().maintenanceRecord(this.mResetMileage, this.mResetDate);
            }
            LogUtil.logD(TAG, "initMaintenanceInfo stdMtcMileage = " + this.mStdMtcMileage + " stdMtcTime = " + this.mStdMtcTime + " stdMtcAdvMileage = " + this.mStdMtcAdvMileage + " stdMtcAdvTime = " + this.mStdMtcAdvTime + " resetMileage =" + resetMileage + " resetDate = " + resetDate);
            onMaintenanceInfoChangeEvent();
            return;
        }
        this.mResetDate = -1L;
        this.mResetMileage = -1;
        TspManager.getInstance().getMaintenanceInfo();
    }

    public static ComponentName getForegroundActivity() {
        IActivityManager activityManager = ActivityManager.getService();
        try {
            List<ActivityManager.StackInfo> stackInfoList = activityManager.getAllStackInfos();
            int size = stackInfoList.size();
            for (int index = 0; index < size; index++) {
                ActivityManager.StackInfo stackInfo = stackInfoList.get(index);
                if (stackInfo.visible) {
                    return stackInfo.topActivity;
                }
            }
            return null;
        } catch (RemoteException e) {
            return null;
        }
    }
