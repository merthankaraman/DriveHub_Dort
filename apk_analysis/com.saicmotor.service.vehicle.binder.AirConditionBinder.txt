    private int getVehicleType() throws RemoteException {
        CarInfoManager carInfoManager = this.mCarInfoManager;
        if (carInfoManager != null) {
            try {
                byte[] carModelArray = carInfoManager.getByteProperty(561005100);
                if (Utils.isEmpty(carModelArray)) {
                    return -1;
                }
                return carModelArray[0] & 255;
            } catch (CarNotConnectedException exception) {
                exception.printStackTrace();
            } catch (IllegalArgumentException ex) {
                ex.printStackTrace();
                return this.DEFAULT_NUMBER;
            }
        }
        return -1;
    }

    private float convertCarServerData(int pickerShowTemp) {
        if (setTempJudgment(pickerShowTemp)) {
            if (isAcTempRangeType16To30()) {
                if (pickerShowTemp >= 31) {
                    return TEMPERATURE_MAX_2;
                }
                if (pickerShowTemp <= 15) {
                    return TEMPERATURE_MIN_2;
                }
            } else {
                if (pickerShowTemp >= 33) {
                    return TEMPERATURE_MAX;
                }
                if (pickerShowTemp <= 17) {
                    return TEMPERATURE_MIN;
                }
            }
            return pickerShowTemp;
        }
        return TEMPERATURE_ERROR_CONST;
    }

    private int convertPickerTemp(float carTempData) {
        if (isAcTempRangeType16To30()) {
            if (carTempData == TEMPERATURE_MAX_2) {
                return 31;
            }
            if (carTempData == TEMPERATURE_MIN_2) {
                return 15;
            }
            if (carTempData > 30.0f || carTempData < 16.0f || !isEvenTemp(carTempData)) {
                LogUtil.logD(TAG, "error set = " + carTempData);
                return -1;
            }
        } else {
            if (carTempData == TEMPERATURE_MAX) {
                return 33;
            }
            if (carTempData == TEMPERATURE_MIN) {
                return 17;
            }
            if (carTempData > 32.0f || carTempData < 18.0f || !isEvenTemp(carTempData)) {
                LogUtil.logD(TAG, "error set = " + carTempData);
                return -1;
            }
        }
        return (int) carTempData;
    }

    private boolean isEvenTemp(float carTempData) {
        LogUtil.logD(TAG, "isEvenTemp carTempData = " + carTempData);
        String tempStr = String.valueOf(carTempData);
        String[] digital = tempStr.split(TEMP_SPLITTER);
        return TEMP_ODD_SIGN.equals(digital[1]);
    }
